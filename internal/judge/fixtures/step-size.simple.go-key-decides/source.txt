package app

import tea "github.com/charmbracelet/bubbletea"

type Model struct {
	longSelected bool
}

func New() Model {
	return Model{}
}

func (m Model) View() string {
	if m.longSelected {
		return "Classic 25m 5m 15m\n> Long 50m 10m 30m"
	}
	return "> Classic 25m 5m 15m\nLong 50m 10m 30m"
}

func (m Model) Init() tea.Cmd {
	return nil
}

func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	m.longSelected = true
	return m, nil
}
