+type Account struct {
+	Balance  float64
+	Currency string
+}
+
+func TransferService(from, to *Account, amount float64, currency string) error {
+	if from.Currency != currency || to.Currency != currency {
+		return errors.New("currency mismatch")
+	}
+	if from.Balance < amount {
+		return errors.New("insufficient funds")
+	}
+	from.Balance -= amount
+	to.Balance += amount
+	return nil
+}
