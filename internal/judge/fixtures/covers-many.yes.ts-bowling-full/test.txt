import { beforeEach, describe, expect, it } from "vitest";
import { Game } from "./game";

describe("bowling", () => {
  let game: Game;
  beforeEach(() => {
    game = new Game();
  });
  const rollMany = (n: number, pins: number) => {
    for (let i = 0; i < n; i++) game.roll(pins);
  };

  it("scores a gutter game as zero", () => {
    rollMany(20, 0);
    expect(game.score()).toBe(0);
  });

  it("scores all ones as twenty", () => {
    rollMany(20, 1);
    expect(game.score()).toBe(20);
  });

  it("adds the next roll to a spare", () => {
    game.roll(5);
    game.roll(5);
    game.roll(3);
    rollMany(17, 0);
    expect(game.score()).toBe(16);
  });

  it("adds the next two rolls to a strike", () => {
    game.roll(10);
    game.roll(3);
    game.roll(4);
    rollMany(16, 0);
    expect(game.score()).toBe(24);
  });

  it("scores a perfect game as 300", () => {
    rollMany(12, 10);
    expect(game.score()).toBe(300);
  });
});
